   --> Operators: this are basically used to perform calculations/ checking the conditions 
          
           --be fore we begin lets understand operators and operands
     
           -- operator : it is symbol defines what   operations  to be performed
               ex: + , - ,/ , % , *
           
            -- operand : An operand is the value or variable on which an operator performs an operation.  
                   ex: a = 10;
                        b = 20;
                       a + b = 30  --  a , b are operands  and + is called as operator

 --> Javascripts Operators : There are 5 types of operators they are
     
          1.Arithematic operator :  + , - , /, % , *
      
          2.Assignment operator / compound operator: += , -= , /=, *=, %=
          
          3.Comparison operator : == , === ,<= , >= , < , > ,!=
                
                 -->  ==   -- double equals used to compare two values 
                 --> === -- triple quals used for exact comparison checks the value and type of the data
                            work same as AND operator
                                 
          4.Logical Operators : &&(AND) , ||(OR), !(NOT)
              
                && -- The both conditions must be true then only output is true
                || -- atleast one condition must be true
                ! -- Boolean reverese we use it for one condition
        
         5 . Increment / Decrement:  (++ / --) : 
            
            Increment(++) :   adding +1 to the original value
                      ex :
                             var  a = 10;
                              a++; --> a + 1 = 10 + 1 = 11 // expected value
                              console.log(a);

            Decrement(--) : Subtracting -1 from the original value
                  
                     ex :
                            var b = 12;
                            b--;    --> b-- = 12 -1 = 11 // expected value
                            console.log(b);
